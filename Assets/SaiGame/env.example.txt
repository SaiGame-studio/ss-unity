# SaiServer / SaiAuth environment configuration
#
# HOW TO USE
# - No .env yet: copy this file to the Unity project root (next to the Assets folder)
#   and rename it to .env (or click "Create .env" on the SaiServer inspector).
# - .env already exists: copy only the keys you need from this file into it.
# - Turn on "Load .env on Play" in the SaiServer inspector (Environment section)
#   so the values below are applied when entering Play mode.
# - .env is git-ignored and must NOT be committed.

# Game ID registered on the Sai Game platform
# Applied only when "Override game id" is checked on SaiServer.
GAME_ID=your-game-id-here

# Auto-login credentials for SaiAuth
# Applied only when "Override credentials" is checked on SaiServer.
USERNAME=your-username-here
PASSWORD=your-password-here
